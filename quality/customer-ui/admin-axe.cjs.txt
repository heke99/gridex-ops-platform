const{chromium}=require('/opt/codex/cua_node/lib/node_modules/playwright');const fs=require('node:fs');(async()=>{const b=await chromium.launch({executablePath:'/usr/bin/chromium',headless:true,args:['--no-sandbox']});const p=await b.newPage();const results=[];for(const viewport of[{width:390,height:844},{width:1280,height:900}]){await p.setViewportSize(viewport);for(const name of['dashboard','queue','support','billing','switches','website','settings','navigation']){await p.goto('http://localhost:4173/'+(name==='navigation'?'navigation?mode=company&pathname=/admin/customers':'admin-fixture/'+name));await p.locator('h1').first().waitFor();await p.evaluate(()=>Promise.all(document.getAnimations().map(a=>a.finished.catch(()=>{}))));await p.addScriptTag({path:'/workspace/gridex-ops-platform/node_modules/axe-core/axe.min.js'});for(const state of ['default','expanded']){if(state==='expanded')await p.locator('details').evaluateAll(es=>es.forEach(e=>e.open=true));const result=await p.evaluate(async()=>await axe.run(document.body,{runOnly:{type:'tag',values:['wcag2a','wcag2aa','wcag21a','wcag21aa']}}));results.push({view:name,viewport,state,width:await p.evaluate(()=>document.documentElement.scrollWidth),violations:result.violations.map(v=>({id:v.id,impact:v.impact,nodes:v.nodes.map(n=>({target:n.target,summary:n.failureSummary}))}))});if(state==='default')await p.screenshot({path:`/tmp/gridex-ui-browser/admin-${name}-${viewport.width}.png`,fullPage:true})}}}fs.writeFileSync('/tmp/gridex-ui-browser/admin-axe-results.json',JSON.stringify(results,null,2));console.log(JSON.stringify({states:results.length,violations:results.filter(r=>r.violations.length),overflow:results.filter(r=>r.width>r.viewport.width)}));await b.close()})().catch(e=>{console.error(e);process.exit(1)});
