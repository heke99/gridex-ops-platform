function serialize(value:any):any{
 if(value==null||typeof value==='boolean'||typeof value==='string'||typeof value==='number')return value
 if(Array.isArray(value))return value.map(serialize)
 if(typeof value==='function')return {__action:value.name||'anonymous'}
 if(React.isValidElement(value)){
  const e=value as any;let type=e.type
  if(typeof type==='symbol')return serialize(e.props.children)
  if(typeof type==='function'&&!known.has(type.name))return serialize(type(e.props))
  if(typeof type==='function')type=type.name
  const props:any={};for(const[key,v]of Object.entries(e.props)){if(key==='prefetch')continue;props[key]=serialize(v)}
  return {__element:type,key:e.key,props}
 }

}
