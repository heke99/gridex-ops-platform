const{chromium}=require('/opt/codex/cua_node/lib/node_modules/playwright');const fs=require('node:fs');(async()=>{const browser=await chromium.launch({executablePath:'/usr/bin/chromium',headless:true,args:['--no-sandbox']});const page=await browser.newPage();const results=[];for(const viewport of[{width:320,height:844},{width:390,height:844},{width:1024,height:900},{width:1280,height:900}]){await page.setViewportSize(viewport);for(const view of['dashboard','queue','support','billing','switches','website','settings']){await page.goto('http://localhost:4173/admin-fixture/'+view+'?shell=1');await page.locator('h1').first().waitFor();await page.evaluate(()=>Promise.all(document.getAnimations().map(a=>a.finished.catch(()=>{}))));for(const state of['default','expanded']){if(state==='expanded')await page.locator('details').evaluateAll(es=>es.forEach(e=>e.open=true));results.push({view,viewport,state,width:await page.evaluate(()=>document.documentElement.scrollWidth),height:await page.evaluate(()=>document.documentElement.scrollHeight)});if(state==='default'&&['switches','dashboard'].includes(view)&&[390,1280].includes(viewport.width))await page.screenshot({path:`/tmp/gridex-ui-browser/final-${view}-${viewport.width}.png`,fullPage:true})}}}fs.writeFileSync('/tmp/gridex-ui-browser/admin-layout-results.json',JSON.stringify(results,null,2));const overflow=results.filter(r=>r.width>r.viewport.width);console.log(JSON.stringify({states:results.length,overflow,heights:results.filter(r=>r.viewport.width===390&&r.state==='default').map(r=>({view:r.view,height:r.height}))}));await browser.close();if(overflow.length)process.exit(1)})().catch(e=>{console.error(e);process.exit(1)});
