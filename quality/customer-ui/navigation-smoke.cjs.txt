const {chromium}=require('/opt/codex/cua_node/lib/node_modules/playwright');
const fs=require('node:fs');
(async()=>{const browser=await chromium.launch({executablePath:'/usr/bin/chromium',headless:true,args:['--no-sandbox']});const results=[];const failures=[];const page=await browser.newPage();page.on('pageerror',e=>failures.push(e.message));async function record(name,fn){await fn();results.push({name,passed:true})}const assert=(v,m)=>{if(!v)throw Error(m)};
await page.setViewportSize({width:1280,height:900});
await page.goto('http://localhost:4173/navigation?pathname=/admin/customers/intake');
await record('desktop active destination is singular and opens the customer group',async()=>{await page.getByRole('navigation',{name:'Huvudnavigation',exact:true}).waitFor();const current=page.locator('[aria-current=page]');assert(await current.count()===1,'multiple active links');assert(await current.getAttribute('href')==='/admin/customers/intake','wrong destination');assert(await current.isVisible(),'active group closed');assert(await page.locator('aside').boundingBox().then(b=>b.width)===260,'sidebar width');});
await record('desktop collapsed groups remain keyboard accessible',async()=>{const nav=page.getByRole('navigation',{name:'Huvudnavigation',exact:true});const summary=nav.locator('summary').filter({hasText:'System & governance'});await summary.focus();await page.keyboard.press('Enter');await nav.getByRole('link',{name:'Systemstatus',exact:true}).waitFor({state:'visible'});await summary.press('Enter');assert(!await nav.getByRole('link',{name:'Systemstatus',exact:true}).isVisible(),'group did not close');});
await page.screenshot({path:'/tmp/gridex-ui-browser/navigation-desktop.png',fullPage:true});
await page.setViewportSize({width:390,height:844});
await record('mobile page selector includes all visible grouped destinations without overflow',async()=>{const select=page.getByLabel('Gå till sida', {exact:true});assert(await select.isVisible(),'mobile selector absent');assert(await select.inputValue()==='/admin/customers/intake','wrong mobile selection');assert(await select.locator('option[value="/admin/work-queue"]').count()===1,'work queue unreachable');assert(await page.evaluate(()=>document.documentElement.scrollWidth===390),'mobile overflow');await select.selectOption('/admin/work-queue');await page.waitForURL('**/admin/work-queue');assert(await select.inputValue()==='/admin/work-queue','route not reflected');});
await record('mobile platform mode retains the existing action payload',async()=>{await page.locator('summary').filter({hasText:'Byt vy'}).click();await page.getByRole('button',{name:'Bolagsvy',exact:true}).click();await page.waitForFunction(()=>window.__navigationSubmission?.mode==='company');const data=await page.evaluate(()=>window.__navigationSubmission);assert(data.company_id==='company-a','company scope lost');});
await page.goto('http://localhost:4173/navigation?mode=company&pathname=/admin/customers');
await record('mobile company selection retains the server action and selected company',async()=>{await page.locator('summary').filter({hasText:'Byt vy'}).click();const select=page.getByLabel('Aktivt bolag');assert(await select.inputValue()==='company-a','current company missing');await select.selectOption('company-b');await page.waitForFunction(()=>window.__navigationSubmission?.company_id==='company-b');assert((await page.evaluate(()=>window.__navigationSubmission)).mode==='company','mode lost');});
await page.goto('http://localhost:4173/navigation?role=tenant&pathname=/admin/customers');
await record('tenant mobile navigation excludes platform and unavailable live destinations',async()=>{const select=page.getByLabel('Gå till sida',{exact:true});assert(await select.isVisible(),'tenant selector absent');assert(await select.locator('option[value="/admin/platform/security"]').count()===0,'platform leak');assert(await select.locator('option[value="/admin/operations/switches"]').count()===0,'live access leak');assert(await select.locator('option[value="/admin/customer-cases"]').count()===1,'permitted support absent');assert(await page.locator('summary').filter({hasText:'Byt vy'}).count()===0,'tenant has platform toggle');assert(await page.evaluate(()=>document.documentElement.scrollWidth===390),'tenant overflow');});
await page.screenshot({path:'/tmp/gridex-ui-browser/navigation-mobile.png',fullPage:true});
assert(failures.length===0,failures.join('\n'));fs.writeFileSync('/tmp/gridex-ui-browser/navigation-results.json',JSON.stringify({scenarios:results,uncaughtErrors:failures},null,2));console.log(JSON.stringify({passed:results.length,uncaughtErrors:failures.length}));await browser.close()})().catch(e=>{console.error(e);process.exit(1)});
