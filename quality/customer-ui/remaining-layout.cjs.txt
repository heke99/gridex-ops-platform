const{chromium}=require('/opt/codex/cua_node/lib/node_modules/playwright');const fs=require('node:fs');
(async()=>{const b=await chromium.launch({executablePath:'/usr/bin/chromium',headless:true,args:['--no-sandbox','--disable-dev-shm-usage','--renderer-process-limit=1']});const p=await b.newPage();const rows=[];for(const width of[320,390,1024,1280]){await p.setViewportSize({width,height:900});for(const view of['info','facility','metering','outbound','messages','events','pricing','campaigns','users','integrations','info-readonly','integrations-ready','events-platform','messages-platform']){await p.goto('http://localhost:4173/admin-fixture/'+view+'?shell=1');await p.locator('h1').first().waitFor();await p.evaluate(()=>Promise.all(document.getAnimations().map(a=>a.finished.catch(()=>{}))));for(const state of['default','expanded']){if(state==='expanded')await p.locator('details').evaluateAll(xs=>xs.forEach(x=>x.open=true));const overflow=await p.evaluate(()=>({width:document.documentElement.scrollWidth,items:[...document.querySelectorAll('body *')].filter(x=>x.getBoundingClientRect().right>innerWidth+1).slice(0,8).map(x=>({tag:x.tagName,cls:x.className,text:x.textContent.slice(0,65)}))}));rows.push({view,viewport:width,state,...overflow});if(width===390&&state==='default')await p.screenshot({path:'/tmp/gridex-ui-browser/remaining-'+view+'-390.png',fullPage:true})}}}fs.writeFileSync('/tmp/gridex-ui-browser/remaining-layout-results.json',JSON.stringify(rows,null,2));console.log(JSON.stringify({states:rows.length,overflow:rows.filter(x=>x.width>x.viewport)}));await b.close()})().catch(e=>{console.error(e);process.exit(1)});
